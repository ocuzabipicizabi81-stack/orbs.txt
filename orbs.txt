    function waitForWebpack(callback) {
        const checkInterval = 100;
        const maxAttempts = 100;
        let attempts = 0;

        const check = () => {
            if (attempts >= maxAttempts) {
                console.error("[Discord Quest Assistant] Webpack zaman aşımına uğradı.");
                alert("Discord modülleri yüklenemedi. Lütfen sayfayı yenileyin.");
                return;
            }

            if (typeof window.webpackChunkdiscord_app === 'undefined') {
                attempts++;
                setTimeout(check, checkInterval);
                return;
            }

            try {
                const originalJQuery = window.$;
                delete window.$;

                const webpackRequire = window.webpackChunkdiscord_app.push([[Symbol()], {}, (require) => require]);
                window.webpackChunkdiscord_app.pop();

                if (originalJQuery) window.$ = originalJQuery;

                if (!webpackRequire || !webpackRequire.c || Object.keys(webpackRequire.c).length < 10) {
                    attempts++;
                    setTimeout(check, checkInterval);
                    return;
                }

                callback(webpackRequire);
            } catch (error) {
                attempts++;
                setTimeout(check, checkInterval);
            }
        };

        check();
    }

    function findModule(webpackRequire, filter) {
        if (!webpackRequire || !webpackRequire.c) return null;
        for (const module of Object.values(webpackRequire.c)) {
            if (module?.exports) {
                const exports = module.exports;
                if (exports.A && filter(exports.A)) return exports.A;
                if (exports.Ay && filter(exports.Ay)) return exports.Ay;
                if (exports.ZP && filter(exports.ZP)) return exports.ZP;
                if (filter(exports)) return exports;
            }
        }
        return null;
    }

    function loadStores(webpackRequire) {
        try {
            const QuestsStore = findModule(webpackRequire, m => m.__proto__?.getQuest);
            const ChannelStore = findModule(webpackRequire, m => m.__proto__?.getAllThreadsForParent);
            const GuildChannelStore = findModule(webpackRequire, m => m.getSFWDefaultChannel);
            const api = findModule(webpackRequire, m => m.Bo?.get || m.tn?.get);

            if (!QuestsStore || !api) return null;

            return { QuestsStore, ChannelStore, GuildChannelStore, api: api.Bo || api.tn || api };
        } catch (error) {
            return null;
        }
    }

            const isEnrolled = !!quest.userStatus?.enrolledAt;
            const taskConfig = quest.config.taskConfig ?? quest.config.taskConfigV2;
            if (!taskConfig || !taskConfig.tasks) return false;
            const hasSupportedTask = supportedTasks.some(type => taskConfig.tasks[type] !== null && taskConfig.tasks[type] !== undefined);

            return isEnrolled && !isCompleted && !isExpired && hasSupportedTask;
        });

            taskType: taskType || "GÖREV",
            secondsNeeded,
            currentProgress,
            completed: currentProgress >= secondsNeeded,
            enrolledAt: new Date(quest.userStatus.enrolledAt).getTime(),
            questName: quest.config.messages?.questName ?? quest.config.application?.name ?? "Discord Görevi"

        if (countElem) countElem.innerText = `📊 Aktif: ${uncompletedCount} / ${questStates.length}`;

            if (totalSeconds <= 0) {
                timeElem.innerText = "⏱️ Toplam Süre: Tamamlandı 🎉";
            } else {
                const mins = Math.floor(totalSeconds / 60);
                const secs = Math.floor(totalSeconds % 60);

        currentQuestStates.forEach(state => {
            const { quest } = state;
            const pct = Math.min(100, Math.floor((state.currentProgress / state.secondsNeeded) * 100));

            const card = document.createElement("div");
            card.className = "dqu-quest-card";
            card.id = `dqu-card-${quest.id}`;
            card.innerHTML = `
                <div class="dqu-card-header">
                    <div class="dqu-quest-title">${state.questName}</div>
                    <div class="dqu-card-actions">
                        <div class="dqu-status-badge badge-pending" id="dqu-badge-${quest.id}">${state.completed ? 'Tamamlandı' : 'Bekliyor'}</div>
                    </div>
                </div>
                <div class="dqu-quest-type">
                    <span>📌 Tür: ${state.taskType}</span>
                    <span id="dqu-eta-${quest.id}" class="dqu-live-eta"></span>
                </div>
                <div class="dqu-progress-bg">
                    <div class="dqu-progress-bar" id="dqu-bar-${quest.id}" style="width: ${pct}%"></div>
                </div>
                <div class="dqu-progress-info">
                    <span id="dqu-text-${quest.id}">${Math.floor(state.currentProgress)}s / ${state.secondsNeeded}s</span>
                    <span id="dqu-pct-${quest.id}">${pct}%</span>
                </div>
            `;
            container.appendChild(card);
        });
        log(`${activeQuests.length} adet kabul edilmiş yapılabilir görev tespit edildi.`, "info");
    };

    // SANİYE SANİYE CANLI EKRAN İLERLEMESİ GÜNCELLEYİCİ
    function startLiveTicker() {
        if (window.dquLiveInterval) clearInterval(window.dquLiveInterval);
        window.dquLiveInterval = setInterval(() => {
            if (!isRunning || !currentQuestStates || currentQuestStates.length === 0) return;

            currentQuestStates.forEach(state => {
                if (!state.completed && state.currentProgress < state.secondsNeeded) {
                    state.currentProgress = Math.min(state.secondsNeeded, state.currentProgress + 1);
                    const rem = Math.max(0, Math.ceil(state.secondsNeeded - state.currentProgress));
                    const isVideo = state.taskType.includes("VIDEO") || state.taskType.includes("WATCH");
                    const etaText = isVideo ? `⏱ Kalan: ${rem}s` : `⏱ Tahmini: ${Math.ceil(rem / 60)} dk`;
                    updateCardProgress(state.quest.id, state.currentProgress, state.secondsNeeded, false, etaText);
                }
            });

            calculateTotalTime(currentQuestStates);
        }, 1000);
    }

    // EKLENTİ İLE 100% UYUMLU STEP FONKSİYONLARI (VİDEO + İZLEME + MOBİL DESTEKLİ)
    async function processVideoStep(state, api) {
        const { quest, secondsNeeded } = state;
        const speed = 1;
        const nextTime = Math.min(secondsNeeded, state.currentProgress + speed + Math.random());

        try {
            const res = await api.post({ url: `/quests/${quest.id}/video-progress`, body: { timestamp: nextTime } });
            state.currentProgress = nextTime;
            const rem = Math.max(0, Math.ceil(secondsNeeded - state.currentProgress));
            updateCardProgress(quest.id, state.currentProgress, secondsNeeded, false, `⏱ Kalan: ${rem}s`);
            log(`Video İlerlemesi: ${Math.floor(state.currentProgress)} / ${secondsNeeded} sn`, "info");

            if (res.body.completed_at !== null || state.currentProgress >= secondsNeeded) {
                state.completed = true;
                updateCardProgress(quest.id, secondsNeeded, secondsNeeded, true, "");
                await api.post({ url: `/quests/${quest.id}/video-progress`, body: { timestamp: secondsNeeded } });
                log(`✨ ${state.questName} Tamamlandı!`, "success");
                if (userSettings.autoClaimOrbs) await claimQuestOrbs(quest, api);
            }
        } catch (error) {}
    }

    async function processHeartbeatStep(state, stores) {
        const { api, ChannelStore, GuildChannelStore } = stores;
        const { quest, taskType, secondsNeeded } = state;

        let channelId = ChannelStore?.getSortedPrivateChannels()[0]?.id;
        if (!channelId && GuildChannelStore) {
            const guilds = Object.values(GuildChannelStore.getAllGuilds());
            const voice = guilds.find(g => g?.VOCAL?.length > 0);
            if (voice) channelId = voice.VOCAL[0].channel.id;
        }

        const streamKey = channelId ? `call:${channelId}:1` : `call:${quest.id}:1`;

        try {
            const response = await api.post({
                url: `/quests/${quest.id}/heartbeat`,
                body: { stream_key: streamKey, terminal: false }
            });

            const serverProgress = response.body?.progress?.[taskType]?.value ?? state.currentProgress;
            state.currentProgress = serverProgress;

            const rem = Math.max(0, Math.ceil((secondsNeeded - state.currentProgress) / 60));
            updateCardProgress(quest.id, state.currentProgress, secondsNeeded, state.currentProgress >= secondsNeeded, `⏱ Tahmini: ${rem} dk`);
            log(`Görev İlerlemesi: ${Math.floor(state.currentProgress)} / ${secondsNeeded} sn`, "info");

            if (state.currentProgress >= secondsNeeded) {
                await api.post({
                    url: `/quests/${quest.id}/heartbeat`,
                    body: { stream_key: streamKey, terminal: true }
                });
                state.completed = true;
                updateCardProgress(quest.id, secondsNeeded, secondsNeeded, true, "");
                log(`✨ ${state.questName} Tamamlandı!`, "success");
                if (userSettings.autoClaimOrbs) await claimQuestOrbs(quest, api);
            }
        } catch (error) {}
    }

    async function claimQuestOrbs(quest, api) {
        log(`🔮 Ödül / Orbs Talep Ediliyor: ${quest.config.messages?.questName ?? 'Görev'}`, "info");
        try {
            const res = await api.post({ url: `/quests/${quest.id}/claim`, body: { platform: 0 } });
            if (res.body) log(`🎉 Ödül/Orbs Başarıyla Toplandı!`, "success");
        } catch(e) {}
    }

    // ANA ÇALIŞTIRMA KONTROLCÜSÜ (Eklentideki `runQuestCode` Mantığı)
    async function runQuestEngine(stores) {
        const activeQuests = getActiveQuests(stores.QuestsStore);
        if (activeQuests.length === 0) {
            log("Tamamlanacak kabul edilmiş aktif görev bulunamadı!", "success");
            const startBtn = document.getElementById("dqu-start-btn");
            const statusDot = document.getElementById("dqu-dot");
            if (startBtn) {
                startBtn.innerText = "✓ Tüm Görevler Tamamlandı";
                startBtn.className = "dqu-btn dqu-btn-primary dqu-btn-disabled";
            }
            if (statusDot) statusDot.classList.add("stopped");
            isRunning = false;
            return;
        }

        const questStates = activeQuests.map(quest => initializeQuestState(quest));
        currentQuestStates = questStates;

        startLiveTicker();

        const runStateLoop = async (state) => {
            while (!state.completed && isRunning) {
                const isVideo = state.taskType.includes("VIDEO") || state.taskType.includes("WATCH");

                if (isVideo) {
                    await processVideoStep(state, stores.api);
                    if (!state.completed && isRunning) {
                        await new Promise(r => setTimeout(r, 1000 + (Math.random() * 500)));
                    }
                } else {
                    await processHeartbeatStep(state, stores);
                    if (!state.completed && isRunning) {
                        await new Promise(r => setTimeout(r, 20000 + (Math.random() * 2000)));
                    }
                }
            }
        };

        if (userSettings.executionMode === "parallel") {
            log(`⚡ Eşzamanlı Mod: ${questStates.length} görev aynı anda çalıştırılıyor...`, "warn");
            await Promise.all(questStates.map(state => runStateLoop(state)));
        } else {
            for (const state of questStates) {
                if (!isRunning) break;
                if (!state.completed) await runStateLoop(state);
            }
        }

        if (isRunning) {
            log("Tüm görevler başarıyla tamamlandı!", "success");
            const startBtn = document.getElementById("dqu-start-btn");
            const statusDot = document.getElementById("dqu-dot");
            if (startBtn) {
                startBtn.innerText = "✓ Tüm Görevler Tamamlandı";
                startBtn.className = "dqu-btn dqu-btn-primary dqu-btn-disabled";
            }
            if (statusDot) statusDot.classList.add("stopped");
            isRunning = false;
        }
    }

    // WEBPACK HAZIR OLDUĞUNDA BAŞLAT
    waitForWebpack((webpackRequire) => {
        const stores = loadStores(webpackRequire);
        if (!stores) {
            alert("Discord Mağaza/API modülleri bulunamadı. Lütfen Discord'u yenileyin.");
            return;
        }

        currentStores = stores;
        injectUI();
        renderQuests();

        const startBtn = document.getElementById("dqu-start-btn");
        const statusDot = document.getElementById("dqu-dot");

        startBtn.onclick = () => {
            if (isRunning) {
                isRunning = false;
                if (window.dquLiveInterval) { clearInterval(window.dquLiveInterval); window.dquLiveInterval = null; }
                startBtn.innerText = "⚡ Görevleri Otomatik Başlat";
                startBtn.className = "dqu-btn dqu-btn-primary";
                statusDot.classList.add("stopped");
                log("⏸️ İşlem durduruldu.", "warn");
            } else {
                isRunning = true;
                startBtn.innerText = "⏸️ İşlemi Durdur";
                startBtn.className = "dqu-btn dqu-btn-stop";
                statusDot.classList.remove("stopped");
                log("▶ İşlem başlatıldı.", "info");
                runQuestEngine(stores);
            }
        };

        console.log("%c[Discord Quest Assistant] Arayüz Ekranın Sağ Altına Başarıyla Yüklendi!", "color: #34d399; font-size: 14px; font-weight: bold;");
    });
})();
